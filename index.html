        function toggleBillStatus(id) {

            const key = getMonthKey(currentDate.year, currentDate.month);

            const bills = billsData[key];

            const bill = bills.find(b => b.id === id);

            if (bill) {

                if (bill.status === 'paid') {

                    bill.status = 'pending';

                    bill.paymentDate = '';

                } else {

                    bill.status = 'paid';

                    const today = new Date().toISOString().split('T')[0];

                    bill.paymentDate = today;

                }

                renderAll();

                showToast(bill.status === 'paid' ? 'Conta marcada como paga!' : 'Conta marcada como pendente.');

            }

        }

        function handleBillSubmit(event) {

            event.preventDefault();

            const id = document.getElementById('billId').value;

            const name = document.getElementById('billName').value;

            const amount = parseFloat(document.getElementById('billAmount').value);

            const dueDate = parseInt(document.getElementById('billDueDate').value);

            const status = document.getElementById('billStatus').value;

            const payer = document.getElementById('billPayer').value;

            const paymentDate = document.getElementById('billPaymentDate').value;


            const key = getMonthKey(currentDate.year, currentDate.month);


            if (id) {

                const bill = billsData[key].find(b => b.id === id);

                if (bill) {

                    bill.name = name;

                    bill.amount = amount;

                    bill.dueDate = dueDate;

                    bill.status = status;

                    bill.payer = payer;

                    bill.paymentDate = paymentDate;

                }

                showToast('Conta atualizada com sucesso!');

            } else {

                const newBill = {

                    id: 'bill_' + Math.random().toString(36).substr(2, 9),

                    name,

                    amount,

                    dueDate,

                    status,

                    payer,

                    paymentDate: status === 'paid' && !paymentDate ? new Date().toISOString().split('T')[0] : paymentDate

                };

                if (!billsData[key]) billsData[key] = [];

                billsData[key].push(newBill);

                showToast('Nova conta adicionada!');

            }


            closeBillModal();

            renderAll();

        }

        function formatCurrency(value) {

        function formatDateShort(dateStr) {

            if (!dateStr) return '';

            const parts = dateStr.split('-');

            if (parts.length === 3) {

                return `${parts[2]}/${parts[1]}`;

            }

            return dateStr;

        }


        function escapeHtml(str) {

            return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");

        }
